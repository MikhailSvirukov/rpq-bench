cmake_minimum_required(VERSION 3.23)
project(RpqBenchmark LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(BUILD_TESTING OFF CACHE BOOL "Build dependency tests" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "Build shared libraries" FORCE)
set(BUILD_STATIC_LIBS OFF CACHE BOOL "Build static libraries" FORCE)
foreach(dependency GraphBLAS LAGraph)
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${dependency}/CMakeLists.txt")
        message(FATAL_ERROR "Missing ${dependency}: run git submodule update --init --recursive")
    endif()
endforeach()
add_subdirectory(GraphBLAS EXCLUDE_FROM_ALL)
# LAGraph's SuiteSparse integration uses the GraphBLAS targets above directly.
set(SUITESPARSE_ROOT_CMAKELISTS ON)
add_subdirectory(LAGraph EXCLUDE_FROM_ALL)
add_executable(rpq_paths_benchmark src/rpq_paths_benchmark.c)
target_link_libraries(rpq_paths_benchmark PRIVATE LAGraphX LAGraph GraphBLAS::GraphBLAS)
find_package(Git REQUIRED)
foreach(dependency LAGraph GraphBLAS)
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${CMAKE_CURRENT_SOURCE_DIR}/${dependency}" rev-parse HEAD
        OUTPUT_VARIABLE ${dependency}_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
endforeach()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/build-info.json" CONTENT
    "{\"lagraph_commit\":\"${LAGraph_COMMIT}\",\"graphblas_commit\":\"${GraphBLAS_COMMIT}\",\"build_type\":\"${CMAKE_BUILD_TYPE}\",\"graphblas_compact\":\"${GRAPHBLAS_COMPACT}\"}\n")
